# helper functions


def resample(
    original_times,
    data,
    updated_times,
):
    """
    Resample an array onto a new set of time points
    """
    original_times = np.asarray(
        original_times,
        float,
    )
    data = np.asarray(
        data,
        float,
    )
    updated_times = np.asarray(
        updated_times,
        float,
    )

    # a single array vs. a stack of arrays, one per slider position
    if data.ndim == 1:
        resampled = np.interp(
            updated_times,
            original_times,
            data,
        )
    else:
        resampled = np.stack(
            [np.interp(updated_times, original_times, row) for row in data],
        )

    return resampled


def build_slider_payload(
    x,
    series,
    *,
    slider_values,
    slider_name,
    slider_label,
    xlabel,
    ylabel,
    ylabel_template="{label} x {value}",
    title=None,
    initial=0,
    tick_labels=None,
    symmetric_y=True,
    zero_line=True,
    y_domain=None,
    width=720,
    height=360,
):
    """Collect everything the figure needs into one package."""
    x_values = np.asarray(
        x,
        float,
    ).tolist()
    n_points = len(x_values)
    n_frames = len(slider_values)

    # make sure every line matches the time points and the slider
    prepared_series = []
    series_arrays = []
    for s in series:
        data = np.asarray(
            s["data"],
            float,
        )
        if data.ndim == 2 and data.shape[0] != n_frames:
            raise ValueError(
                f"{s['label']}: {data.shape[0]} frames != {n_frames} slider values",
            )
        if data.shape[-1] != n_points:
            raise ValueError(
                f"{s['label']}: {data.shape[-1]} points != {n_points} x values",
            )
        series_arrays.append(data)
        prepared_series.append(
            {
                **s,
                "data": data.tolist(),
            },
        )

    # make sure the slider settings make sense
    if not 0 <= initial < n_frames:
        raise ValueError(f"initial={initial} out of range")
    if tick_labels is not None and len(tick_labels) != n_frames:
        raise ValueError("tick_labels length must match slider_values")

    # fill in the slider labels if none were given
    if tick_labels is None:
        tick_labels = [str(v) for v in slider_values]

    # determine the y axis label at each slider position
    y_labels = [
        ylabel_template.format(
            label=ylabel,
            value=tick,
        )
        for tick in tick_labels
    ]

    # determine the y axis range at each slider position
    if y_domain is not None:
        y_domains = [list(y_domain) for _ in range(n_frames)]
    else:
        y_domains = []
        for idx in range(n_frames):
            combined = np.concatenate(
                [arr[idx] if arr.ndim == 2 else arr for arr in series_arrays],
            )
            if symmetric_y:
                max_abs = (float(np.max(np.abs(combined))) or 1.0) * 1.05
                y_domains.append([-max_abs, max_abs])
            else:
                y_domains.append([float(np.min(combined)), float(np.max(combined))])

    payload = {
        "title": title,
        "width": width,
        "height": height,
        "x": {
            "label": xlabel,
            "values": x_values,
        },
        "y": {
            "labels": y_labels,
            "zero_line": zero_line,
            "domains": y_domains,
        },
        "slider": {
            "name": slider_name,
            "label": slider_label,
            "tick_labels": tick_labels,
            "initial": initial,
        },
        "series": prepared_series,
    }

    return payload


def round_floats(
    obj,
    sig,
):
    """
    Go through floats, lists, and dicts and round any floats
    """
    if isinstance(obj, float):
        rounded = float(f"{obj:.{sig}g}")
    elif isinstance(obj, list):
        rounded = [round_floats(item, sig) for item in obj]
    elif isinstance(obj, dict):
        rounded = {key: round_floats(value, sig) for key, value in obj.items()}
    else:
        rounded = obj

    return rounded


def write_widget_js(
    payload,
    out_path,
    sig=None,
):
    """
    Write the payload to a ".js" file that Textbook markdown pages can load via a
    <div data-src=""> element
    """
    out_path = Path(out_path)

    # optionally round values to keep the filesize small
    if (not isinstance(sig, bool)) and (isinstance(sig, int)) and (sig >= 0):
        payload = round_floats(
            payload,
            sig,
        )
    json_payload = json.dumps(
        payload,
        separators=(",", ":"),
    )

    # save the payload as a javascript file that registers the data under the filename
    # - this allows us to import the script with html using (for example):
    #   `<div class="hnn-slider" data-src="scaling_widget.js">`
    out_path.write_text(
        "window.HNN_WIDGET_DATA = window.HNN_WIDGET_DATA || {};\n"
        f'window.HNN_WIDGET_DATA["{out_path.name}"] = {json_payload};\n',
        encoding="utf-8",
    )


def count_drive_spikes(
    cell_response,
    drive_groups,
):
    """Count the spikes from each group of drives in each time bin."""
    spike_times = np.concatenate(cell_response.spike_times)
    spike_types = np.concatenate(cell_response.spike_types).astype(str)
    bin_edges = np.linspace(
        0,
        cell_response.times[-1],
        50,
    )

    # count the spikes in each time bin for each group of drives
    counts = {}
    for label, prefixes in drive_groups.items():
        in_group = np.zeros(
            len(spike_types),
            dtype=bool,
        )
        for prefix in prefixes:
            in_group |= np.char.startswith(
                spike_types,
                prefix,
            )
        group_counts, _ = np.histogram(
            spike_times[in_group],
            bin_edges,
        )
        counts[label] = group_counts

    histogram = {
        "bin_edges": bin_edges,
        "counts": counts,
    }

    return histogram


def padded_peaks(
    peaks,
    dynamic,
):
    """Give the top of the y-axis for each pair of slider positions."""
    peaks = np.asarray(
        peaks,
        float,
    )

    # use the overall peak everywhere unless the axis follows the sliders
    if not dynamic:
        peaks = np.full_like(
            peaks,
            peaks.max(),
        )

    # leave a little room above the peak, and avoid a zero-height axis
    tops = np.where(peaks > 0, peaks, 1.0) * 1.05

    return tops


def get_slider_shape(
    sliders,
):
    """Count the positions of each slider, and check each starting position exists."""
    # get the length of every slider
    # ------------------------------
    slider_shape = tuple(len(slider["tick_labels"]) for slider in sliders)

    # sanity checks
    # ------------------------------
    for slider, n_positions in zip(
        sliders,
        slider_shape,
        strict=True,
    ):
        if not 0 <= slider.get("initial", 0) < n_positions:
            print(f"{slider['label']}: initial position is out of range")
            print("\nResetting initial position to 0")
            slider["initial"] = 0

    return slider_shape


def build_histogram_panel(
    *,
    bin_edges,
    counts,
    colors,
    inverted_groups,
    slider_shape,
    title,
    ylabel,
):
    # set up
    # ------------------------------
    # cast histogram bin_edges to an array
    bin_edges = np.asarray(
        bin_edges,
        float,
    )

    # get histogram bin counts
    n_bins = len(bin_edges) - 1

    # `counts` maps each *grouped* drive label ("Proximal", "Distal") to an array
    # whose dimensions are (*slider_shape, n_bins), with one dimension per slider
    #
    # for example: if slider 1 (i) has 5 positions and slider 2 (j) has 4 positions,
    # each array must have shape (i=5, j=4, n_bins). `counts[i, j]` returns the bins
    # (spike counts) for slider 1 at index `i` and slider 2 at index `j`
    #
    # this grid is *always* necessary because the "front end" (javascript) does not
    # know whether any, some, or all of the slider parameters will affect the histogram
    # output of the simulation, and so the histogram data are needed for every
    # slider permutation
    groups = []
    for label, group_counts in counts.items():
        group_counts = np.asarray(group_counts)
        if group_counts.shape != slider_shape + (n_bins,):
            raise ValueError(f"{label}: counts shape {group_counts.shape} does not fit")
        groups.append(
            {
                "label": label,
                "color": to_hex(colors[label]),
                "inverted": label in inverted_groups,
                "counts": group_counts.tolist(),
            },
        )

    # prepare panel
    # ------------------------------
    # keep the y-axis the same for every slider position so the bars can be compared
    tallest = max(np.max(group_counts) for group_counts in counts.values())

    panel = {
        "title": title,
        "ylabel": ylabel,
        "bin_edges": bin_edges.tolist(),
        "domain": [0, float(padded_peaks(tallest, False))],
        "groups": groups,
    }

    return panel


def build_dipole_panel(
    *,
    times,
    simulated,
    experimental,
    rmse_labels,
    slider_shape,
    dynamic_y,
    ylabel,
    simulated_label="Simulation",
    simulated_color="#1f77b4",
    experimental_label="Experimental data",
    experimental_color="#ff7f0e",
):
    """Collect the dipole lines and RMSE text into one package."""
    # set up
    # ------------------------------
    # cast times and dipoles to arrays
    times = np.asarray(
        times,
        float,
    )
    simulated = np.asarray(
        simulated,
        float,
    )
    experimental = np.asarray(
        experimental,
        float,
    )

    # `simulated` has dimensions (*slider_shape, n_times): one dipole for every
    # combination of slider positions, because the simulation is re-run for each one.
    # `experimental` is a single dipole, since it never changes with the sliders

    # sanity checks
    # ------------------------------
    if simulated.shape != slider_shape + (len(times),):
        raise ValueError(f"simulated shape {simulated.shape} does not fit")
    if experimental.shape != times.shape:
        raise ValueError("experimental data must match the time points")
    if np.asarray(rmse_labels).shape != slider_shape:
        raise ValueError("rmse_labels must have one entry per combination of sliders")

    # prepare panel
    # ------------------------------
    # find the tallest value for every combination of slider positions
    peaks = np.maximum(
        np.abs(simulated).max(axis=-1),
        np.abs(experimental).max(),
    )
    tops = padded_peaks(
        peaks,
        dynamic_y,
    )

    # center the y-axis on zero, with one range per combination of slider positions
    domains = np.stack(
        [-tops, tops],
        axis=-1,
    )

    panel = {
        "ylabel": ylabel,
        "times": times.tolist(),
        "domains": domains.tolist(),
        "simulated": {
            "label": simulated_label,
            "color": simulated_color,
            "frames": simulated.tolist(),
        },
        "experimental": {
            "label": experimental_label,
            "color": experimental_color,
            "data": experimental.tolist(),
        },
        "rmse": np.asarray(rmse_labels).tolist(),
    }

    return panel

# Interactive plot for adjusting drive mu and sigma

# set simulation parameters
# ---------------------------------------
outname = "TEST_widget"

save_js = True
run_simulations = False

sim_output_path = out_dir / f"{outname}_data.pkl"
drive_name = "evdist1"
dipole_scaling = 3000
dipole_smoothing = 30

param_1_sim_data = {
    "name": "Mean arrival time",
    "param": "mu",
    "initial_index": 0,
    "values": [
        49,
        56,
        # 63,
        # 70,
        # 77,
    ],
}
param_2_sim_data = {
    "name": "Std. deviation",
    "param": "sigma",
    "initial_index": 0,
    "values": [
        1,
        2,
        # 3,
        # 4,
        # 5,
    ]
}

# run simulation loop
# ----------------------------------------

if run_simulations:
    sim_net = neymotin_2020_model()
    add_erp_drives_to_jones_model(sim_net)

    raw_runs = []
    with MPIBackend(ncores):
        for param_1, param_2 in product(
            param_1_sim_data["values"],
            param_2_sim_data["values"],
        ):
            # NOTE: these param references are custom to param_1 and param_2
            drive_net = deepcopy(sim_net)
            drive_net.external_drives[drive_name]["dynamics"]["mu"] = param_1
            drive_net.external_drives[drive_name]["dynamics"]["sigma"] = param_2

            dipoles = simulate_dipole(
                net=drive_net,
                tstop=tstop,
                n_trials=n_trials,
                dt=dt,
            )

            raw_runs.append(
                {
                    "param_1": param_1,
                    "param_2": param_2,
                    "dipole": dipoles[0],
                    "cell_response": drive_net.cell_response,
                },
            )

    sim_output_data = {
        "param_1": param_1_sim_data,
        "param_2": param_2_sim_data,
        "runs": raw_runs,
    }
    with open(sim_output_path, "wb") as f:
        pickle.dump(
            sim_output_data,
            f,
        )
else:
    with open(sim_output_path, "rb") as f:
        sim_output_data = pickle.load(f)

# read the data from the stored simulation output
# ----------------------------------------

drive_groups = {
    "Proximal": ["evprox"],
    "Distal": ["evdist"],
}

sliders = [
    {
        "label": sim_output_data["param_1"]["name"],
        "tick_labels": [
            str(val) for val in sim_output_data["param_1"]["values"]
        ],
        "initial": sim_output_data["param_1"]["initial_index"],
    },
    {
        "label": sim_output_data["param_2"]["name"],
        "tick_labels": [
            str(val) for val in sim_output_data["param_2"]["values"]
        ],
        "initial": sim_output_data["param_2"]["initial_index"],
    },
]
slider_shape = get_slider_shape(sliders)

# process the simulation outputs
# ----------------------------------------
flat_counts = {label: [] for label in drive_groups}
flat_dipoles = []
rmse_labels = []

# NOTE: the runs are saved in param 1 x param 2 order,
# which the reshape below depends on
for run in sim_output_data["runs"]:
    # scale and smooth a copy, so the stored dipole is never changed
    processed_dipole = (
        deepcopy(run["dipole"]).scale(dipole_scaling).smooth(dipole_smoothing)
    )

    # count the drive spikes in each time bin
    histogram = count_drive_spikes(
        run["cell_response"],
        drive_groups,
    )
    for label in drive_groups:
        flat_counts[label].append(histogram["counts"][label])

    # compare to the experimental data, then use fewer time points
    rmse = _rmse(
        processed_dipole,
        threshold_experimental_dpl,
        0.0,
        tstop,
    )
    rmse_labels.append(f"RMSE: {rmse:.4f}")
    flat_dipoles.append(
        resample(
            processed_dipole.times,
            processed_dipole.data["agg"],
            experimental_times,
        ),
    )

drive_counts = {
    label: np.asarray(counts).reshape(
        *slider_shape,
        -1,
    )
    for label, counts in flat_counts.items()
}
simulated = np.asarray(flat_dipoles).reshape(
    *slider_shape,
    -1,
)
rmse_labels = np.asarray(rmse_labels).reshape(slider_shape)

hist_panel = build_histogram_panel(
    bin_edges=histogram["bin_edges"],
    counts=drive_counts,
    colors={
        "Proximal": "r",
        "Distal": "g",
    },
    inverted_groups=["Distal"],
    slider_shape=slider_shape,
    title="External Drive Spike Arrival Times",
    ylabel="Counts",
)

dipole_panel = build_dipole_panel(
    times=experimental_times,
    simulated=simulated,
    experimental=experimental_data,
    rmse_labels=rmse_labels,
    slider_shape=slider_shape,
    dynamic_y=True,
    ylabel=f"Dipole (nAm) x {dipole_scaling}",
)

payload = {
    "width": 720,
    "height": 640,
    "x": {
        "label": "Time (ms)",
        "domain": [0, tstop],
    },
    "sliders": sliders,
    "hist": hist_panel,
    "dipole": dipole_panel,
}

if save_js:
    # fall back to the current folder when run as a notebook
    out_dir = Path.cwd()
    if "__file__" in globals():
        out_dir = Path(__file__).parent

    write_widget_js(
        payload,
        out_dir / f"{outname}.js",
    )
